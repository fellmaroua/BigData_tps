// إيجاد الـ _id للـ Thinkpad واسترجاعه
var thinkpad = db.produits.findOne({ nom: "Thinkpad X230" })
print("Thinkpad par ID:")
printjson(db.produits.findOne({ _id: thinkpad._id }))

// المنتجات بسعر أكبر من 13723
print("Produits prix > 13723:")
db.produits.find({ prix: { $gt: 13723 } }).forEach(printjson)

// أول منتج ultrabook
print("Premier produit ultrabook:")
printjson(db.produits.findOne({ ultrabook: true }))

// أول منتج يحتوي اسمه على Macbook
print("Premier produit contenant 'Macbook':")
printjson(db.produits.findOne({ nom: /Macbook/ }))

// المنتجات التي يبدأ اسمها بـ Macbook
print("Produits dont le nom commence par 'Macbook':")
db.produits.find({ nom: /^Macbook/ }).forEach(printjson)


// حذف كل منتجات Apple
db.produits.deleteMany({ fabriquant: "Apple" })
